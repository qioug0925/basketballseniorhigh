function loadGameState() {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return structuredClone(defaults);
    const data = JSON.parse(raw);
    const s = {...structuredClone(defaults), ...data};
    if (!Array.isArray(s.teams) || s.teams.length !== 2) s.teams = [[],[]];
    if (s.playerSide !== 0 && s.playerSide !== 1) s.playerSide = 0;
    if (s.game && s.game.players) {
      Object.values(s.game.players).forEach(p => {
        if (typeof p.oRebounds !== "number") p.oRebounds = 0;
      });
    }
    return s;
  } catch(e) {
    return structuredClone(defaults);
  }
}

function shuffle(arr) {
  const a = [...arr];
  for (let i=a.length-1;i>0;i--) {
    const j = Math.floor(Math.random()*(i+1));
    [a[i],a[j]]=[a[j],a[i]];
  }
  return a;
}

  const count = ids.filter(id => BOND_811_IDS.includes(id)).length;

function teamOf(id) {
  if (state.teams[0].includes(id)) return 0;
  if (state.teams[1].includes(id)) return 1;
  return -1;
}
function assignToTeam(id, teamIndex) {
  const other = 1 - teamIndex;
  state.teams[other] = state.teams[other].filter(x => x !== id);
  if (state.teams[teamIndex].includes(id)) return true;
  if (state.teams[teamIndex].length >= 3) return false;
  state.teams[teamIndex].push(id);
  return true;
}
function removeFromTeams(id) {
  state.teams[0] = state.teams[0].filter(x => x !== id);
  state.teams[1] = state.teams[1].filter(x => x !== id);
}
function cycleAssign(id) {
  const t = teamOf(id);
  if (t === -1) {
    if (state.teams[0].length < 3) assignToTeam(id, 0);
    else if (state.teams[1].length < 3) assignToTeam(id, 1);
    else alert("两队都已满，请先移除一名球员。");
  } else if (t === 0) {
    if (state.teams[1].length >= 3) { alert("橙队已满，请先移除一名球员。"); return; }
    assignToTeam(id, 1);
  } else {
    removeFromTeams(id);
  }
  syncSelected();
}

function teamStrength(ids) {
  return ids.reduce((sum,id) => {
    const p = playerById(id);
    return sum + p.shot*1.1 + p.drive + p.dribble*.55 +
      p.defense*.65 + p.stamina*.35 + p.pass*.3 +
      p.rebound*.35 + p.vertical*.2;
  },0);
}
function pickBalanced(ids) {
  let best = [ids.slice(0,3), ids.slice(3)];
  let bestDiff = Infinity;
  for (let i=0;i<200;i++) {
    const arr = shuffle(ids);
    const a = arr.slice(0,3);
    const b = arr.slice(3);
    const d = Math.abs(teamStrength(a) - teamStrength(b));
    const adjusted = d + Math.random()*1.6;
    if (adjusted < bestDiff) {
      bestDiff = adjusted;
      best = [a,b];
    }
  }
  return best;
}

}

function clamp(n,min,max) { return Math.max(min,Math.min(max,n)); }
function fatigueFactor(p) { return .65 + .35*(p.energy/100); }
function rand(min,max) { return Math.random()*(max-min)+min; }

function bestDefender(teamIndex) {
  return state.teams[teamIndex].map(id => state.game.players[id])
    .sort((a,b) => (b.defense+b.energy/30)-(a.defense+a.energy/30))[0];
}

function teamReboundWeight(teamIndex) {
  const ids = state.teams[teamIndex];
  return ids.reduce((sum, id) => {
    const p = state.game.players[id];
    const base = playerById(id);
    const w = base.rebound + base.height/60 + base.vertical*0.25 + p.energy/100;
    return sum + Math.max(0.4, w);
  }, 0);
}

function topRebounder(teamIndex) {
  const ids = state.teams[teamIndex];
  return ids.map(id => state.game.players[id]).sort((a,b) => {
    const pa = playerById(a.id);
    const pb = playerById(b.id);
    const va = pa.rebound + pa.height/60 + pa.vertical*0.25 + a.energy/100;
    const vb = pb.rebound + pb.height/60 + pb.vertical*0.25 + b.energy/100;
    return vb - va;
  })[0];
}

function resolveRebound(attackTeam, defendTeam) {
  const defense = state.game.defense;
  let offRebBase = 0.28;
  if (defense === "sag") offRebBase -= 0.10;
  if (defense === "double") offRebBase += 0.08;
  if (defense === "pressure") offRebBase += 0.03;
  const offW = teamReboundWeight(attackTeam);
  const defW = teamReboundWeight(defendTeam);
  const ratio = offW / (offW + defW);
  offRebBase += (ratio - 0.5) * 0.4;
  offRebBase = clamp(offRebBase, 0.12, 0.45);
  const isOffReb = Math.random() < offRebBase;
  if (isOffReb) {
    const player = topRebounder(attackTeam);
    return { teamIndex: attackTeam, player, isOffReb: true };
  } else {
    const player = topRebounder(defendTeam);
    return { teamIndex: defendTeam, player, isOffReb: false };
  }
}

function executePlay(action, style) {
  const g = state.game;
  if (!g || g.over) return false;

  const attackTeam = g.offense;
  const defendTeam = 1-attackTeam;
  const attackerId = Number(document.getElementById("attackerSelect").value);
  const attacker = g.players[attackerId];
  const base = playerById(attackerId);
  const teamIds = state.teams[attackTeam];

  const attackBonds = {
    has811:has811Bond(teamIds),
    forcedIso:isForcedIso(attackTeam)
  };

  const teammates = teamIds.filter(id => id!==attackerId).map(id => g.players[id]);
  if (attackBonds.forcedIso) style = "iso";

  const defense = g.defense;
  const defender = bestDefender(defendTeam);
  const dfBase = playerById(defender.id);

  let narration = "", points = 0, made = false, turnover = false, shotType = "";

  const energyFactor = teamBondEnergyFactor(teamIds);
  const cost = {drive:11,mid:8,three:8,pass:6,reset:4,post:12}[action] || 7;

  attacker.energy = clamp(attacker.energy-cost*energyFactor,8,100);
  teammates.forEach(p => {
    p.energy = clamp(p.energy-(action==="reset" ? 2 : 4)*energyFactor,8,100);
  });
  state.teams[defendTeam].forEach(id => {
    g.players[id].energy = clamp(g.players[id].energy-3,8,100);
  });

  const fatigue = fatigueFactor(attacker);
  const defensePenalty = {
    normal:0,
    pressure:action==="drive" ? .07 : .025,
    sag:action==="three" ? -.08 : .035,
    double:action==="pass" || action==="reset" ? -.05 : .13
  }[defense] || 0;

  const bondHitBonus = teamBondHitBonus(teamIds);

  if (action==="reset") {
    const recovery = teammates.slice().sort((a,b) => b.pass-a.pass)[0];
    if (recovery) recovery.energy = clamp(recovery.energy+4,0,100);
    attacker.energy = clamp(attacker.energy+5,0,100);
    narration = `${base.name}示意队友重新拉开空间，蓝领队友调整站位，${base.name}恢复了一点体力。`;
    if (Math.random()<.12) {
      turnover = true;
      narration += `但传球线路出现问题，${dfBase.name}抢断成功！`;
    }
  } else if (action==="pass") {
    let candidates = teammates.slice();
    if (attackBonds.forcedIso) {
      candidates = candidates.filter(p => !isLoneBlockedPass(base.id,p.id));
    }
    if (candidates.length===0) {
      narration = `${base.name}想要传球，但队友之间没有配合意愿，只能自己强攻！`;
      const chance = clamp(
        .15+base.shot*.105*fatigue+bondHitBonus-defensePenalty-rand(0,.13),
        .06,.82
      );
      made = Math.random()<chance;
      points = 2;
      shotType = "被迫单打";
      if (made) {
        attacker.points += 2;
        narration += `${base.name}强投命中，得到2分！`;
      } else {
        narration += `${base.name}强投不中。`;
      }
    } else {
      const receiver = candidates.slice().sort((a,b) => {
        const pa = playerById(a.id);
        const pb = playerById(b.id);
        const scoreA = pa.shot+pa.pass*.25+Math.random()*2;
        const scoreB = pb.shot+pb.pass*.25+Math.random()*2;
        return scoreB-scoreA;
      })[0];
      const recBase = playerById(receiver.id);
      const quality = clamp(
        .36+recBase.shot*.045+recBase.pass*.025+
        (style==="team" ? .08 : 0)+bondHitBonus-defensePenalty-rand(0,.18),
        .08,.78
      );
      if (Math.random()<quality) {
        const threeChance = recBase.shot>=4 ? .45 : .25;
        points = Math.random()<threeChance ? 3 : 2;
        const hitChance = clamp(
          .22+recBase.shot*.095+receiver.energy/500+
          bondHitBonus-defensePenalty+(style==="team" ? .07 : 0),
          .2,.78
        );
        made = Math.random()<hitChance;
        shotType = points===3 ? "空位三分" : "接球终结";
        narration = `${base.name}分球给${recBase.name}，队友跑位创造机会。`;
        if (made) {
          receiver.points += points;
          receiver.assists += 1;
          narration += `${recBase.name}${shotType}命中，得到${points}分！`;
        } else {
          narration += `${recBase.name}${shotType}没能命中。`;
        }
      } else {
        turnover = true;
        attacker.turnovers += 1;
        narration = `${base.name}尝试传球，但${dfBase.name}判断准确，完成抢断！`;
      }
    }
  } else {
    let skill = 0, label = "";
    if (action==="drive") { skill = base.drive*.62+base.dribble*.38; label = "突破上篮"; }
    if (action==="mid")   { skill = base.shot; label = "中距离投篮"; }
    if (action==="three") { skill = base.shot*.85+base.dribble*.15; label = "三分球"; }
    if (action==="post")  { skill = base.drive*.3+base.rebound*.35+base.height/40; label = "内线强攻"; }

    const isThree = action==="three";
    points = isThree ? 3 : 2;

    const styleBonus =
      (base.id===2 && action==="drive") ? .09 :
      (base.id===6 && action==="drive") ? .05 :
      (base.id===7 && action==="three") ? .07 :
      (base.id===9 && action==="mid") ? .06 :
      (base.id===4 && action==="post") ? -.08 : 0;

    const teamBonus = style==="team" ? .035 : 0;
    const blockRisk = action==="drive" || action==="post"
      ? (dfBase.vertical+dfBase.defense)*.012 : 0;

    let chance = .15+skill*.105*fatigue+teamBonus+styleBonus+
      bondHitBonus-defensePenalty-blockRisk-rand(0,.13);

    if (action==="three") chance -= .08;
    if (action==="post") chance += .02;
    chance = clamp(chance,.06,.82);

    const turnoverChance = clamp(
      .045+(defense==="pressure" ? .07 : 0)+
      (defense==="double" ? .12 : 0)+
      (base.dribble<=1 ? .04 : 0),
      .03,.32
    );

    if (Math.random()<turnoverChance) {
      turnover = true;
      attacker.turnovers += 1;
      narration = `${base.name}尝试${label}，但${dfBase.name}压迫到位，进攻失误！`;
    } else {
      made = Math.random()<chance;
      shotType = label;
      if (made) {
        attacker.points += points;
        narration = `${base.name}发动${label}，晃开防守后命中，得到${points}分！`;
      } else {
        narration = `${base.name}选择${label}，${dfBase.name}干扰成功，投篮偏出。`;
      }
    }
  }

  if (made) {
    if (attackTeam===0) g.blue += points;
    else g.orange += points;
  }

  if (turnover) {
    g.offense = defendTeam;
    addLog(narration);
  } else if (!made) {
    const reb = resolveRebound(attackTeam, defendTeam);
    const rebBase = playerById(reb.player.id);
    if (reb.isOffReb) {
      reb.player.oRebounds = (reb.player.oRebounds || 0) + 1;
      reb.player.rebounds += 1;
      addLog(`${narration} ${rebBase.name}抢下进攻篮板，二次进攻机会！`, "oreb");
    } else {
      reb.player.rebounds += 1;
      narration += ` ${rebBase.name}抢下防守篮板。`;
      g.offense = defendTeam;
      addLog(narration);
    }
  } else {
    g.offense = defendTeam;
    addLog(narration);
  }

  Object.values(g.players).forEach(p => {
    p.energy = clamp(p.energy+1,0,100);
  });

  g.turn++;
  checkWinner();
  return true;
}

function play(action) {
  const g = state.game;
  if (!g || g.over) return;
  if (g.offense !== state.playerSide) return;
  if (aiRunning) return;

  undoState = JSON.parse(JSON.stringify(state));
  let style = document.getElementById("offenseStyle").value;
  executePlay(action, style);
  renderGame();
  saveGame();

  if (!state.game.over && state.game.offense !== state.playerSide) {
    maybeRunAI();
  }
}

function aiChooseAction() {
  const g = state.game;
  const attackTeam = g.offense;
  const teamIds = state.teams[attackTeam];
  const style = isForcedIso(attackTeam) ? "iso" : "team";

  const attackerId = teamIds.slice().sort((a,b) => {
    const pa = g.players[a], pb = g.players[b];
    const ba = playerById(a), bb = playerById(b);
    const sa = (ba.shot + ba.drive)*1.0 + pa.energy/40 + Math.random()*2.5;
    const sb = (bb.shot + bb.drive)*1.0 + pb.energy/40 + Math.random()*2.5;
    return sb - sa;
  })[0];
  document.getElementById("attackerSelect").value = String(attackerId);

  const base = playerById(attackerId);
  const defense = g.defense;
  const energy = g.players[attackerId].energy;

  const options = [];
  const addOption = (action, weight) => {
    if (weight > 0) options.push({ action, weight });
  };

  let driveW = base.drive * 1.0;
  if (base.id === 2) driveW += 2.5;
  if (base.id === 6) driveW += 1.5;
  if (defense === "sag") driveW += 1.5;
  addOption("drive", driveW);

  let midW = base.shot * 0.85;
  if (base.id === 9) midW += 2;
  addOption("mid", midW);

  let threeW = 0;
  if (base.shot >= 3) threeW = (base.shot - 2) * 1.4;
  if (base.id === 6 || base.id === 7) threeW += 2;
  if (defense === "sag") threeW -= 1.2;
  addOption("three", threeW);

  let postW = 0;
  if (base.rebound >= 4) postW = base.rebound * 0.8;
  if (base.height >= 183) postW += 1.5;
  if (base.id === 4) postW -= 1.5;
  addOption("post", postW);

  let passW = base.pass * 0.6;
  if (style === "team") passW += 1.5;
  if (isForcedIso(attackTeam)) passW = 0;
  addOption("pass", passW);

  if (energy < 45) addOption("reset", (50 - energy) * 0.1);

  const totalW = options.reduce((s,o) => s + o.weight, 0);
  let r = Math.random() * totalW;
  let chosen = options[0]?.action || "drive";
  for (const o of options) {
    r -= o.weight;
    if (r <= 0) { chosen = o.action; break; }
  }

  return { action: chosen, style };
}

function aiChooseDefense() {
  const g = state.game;
  const defenseTeam = 1 - state.playerSide;
  const aiScore = defenseTeam===0 ? g.blue : g.orange;
  const playerScore = defenseTeam===0 ? g.orange : g.blue;
  const diff = aiScore - playerScore;
  let choices;
  if (diff <= -4) choices = ["pressure","pressure","double","normal"];
  else if (diff >= 6) choices = ["sag","sag","normal","normal"];
  else choices = ["normal","normal","normal","sag","pressure","double"];
  g.defense = choices[Math.floor(Math.random()*choices.length)];
}

function maybeRunAI() {
  const g = state.game;
  if (!g || g.over) return;
  if (g.offense === state.playerSide) {
    aiRunning = false;
    renderGame();
    return;
  }
  if (aiRunning) return;

  aiRunning = true;
  renderGame();

  clearTimeout(aiTimer);
  aiTimer = setTimeout(() => {
    const current = state.game;
    if (!current || current.over) {
      aiRunning = false;
      renderGame();
      return;
    }
    if (current.offense === state.playerSide) {
      aiRunning = false;
      renderGame();
      return;
    }

    aiChooseDefense();
    const { action, style } = aiChooseAction();
    document.getElementById("offenseStyle").value = style;

    undoState = JSON.parse(JSON.stringify(state));
    executePlay(action, style);

    const after = state.game;
    aiRunning = false;

    if (!after || after.over) {
      renderGame();
      return;
    }
    if (after.offense === state.playerSide) {
      renderGame();
      saveGame();
    } else {
      renderGame();
      saveGame();
      maybeRunAI();
    }
  }, 900);
}

function checkWinner() {
  const g = state.game;
  if (g.blue>=21 || g.orange>=21) {
    g.over = true;
    const winner = g.blue>=21 ? "蓝队" : "橙队";
    addLog(`比赛结束！${winner}获胜，最终比分 ${g.blue}:${g.orange}。`);
    state.lastGame = {
      blue:g.blue,
      orange:g.orange,
      winner,
      endedAt:new Date().toISOString(),
      teams:JSON.parse(JSON.stringify(state.teams))
    };
    persist();
    showResult();
  }
}

function showResult() {
  clearTimeout(aiTimer);
  aiRunning = false;
  document.getElementById("setupScreen").classList.add("hidden");
  document.getElementById("gameScreen").classList.add("hidden");
  document.getElementById("resultScreen").classList.remove("hidden");
  const g = state.game;
  const winner = g.blue>=21 ? "蓝队" : "橙队";
  const playerWon = (winner === "蓝队" && state.playerSide === 0) ||
                    (winner === "橙队" && state.playerSide === 1);
  document.getElementById("resultText").innerHTML =
    `<p style="font-size:25px;font-weight:900">${winner}获胜！</p>`+
    `<p style="font-size:17px">${playerWon ? "🎉 你赢了！" : "😅 你输了，再打一场？"}</p>`+
    `<p>最终比分：<b>${g.blue} : ${g.orange}</b></p>`+
    `<p class="muted">比赛记录已自动保存。你可以再打一场，或者返回选人界面。</p>`;
  persist();
}

function undoLast() {
  if (!undoState) {
    alert("目前没有可以撤销的回合。");
    return;
  }
  clearTimeout(aiTimer);
  aiRunning = false;
  state = undoState;
  undoState = null;
  if (state.game.over) showResult();
  else showGame();
  persist();
}

function newMatch() {
  clearTimeout(aiTimer);
  aiRunning = false;
  state.game = null;
  state.logs = [];
  state.teams = [[],[]];
  state.selected = [];
  undoState = null;
  saveGame();
  backToSetup();
}

function exportSave() {
  const text = JSON.stringify(state);
  const area = document.getElementById("saveText");
  area.value = text;
  area.focus();
  area.select();
  try {
    if (navigator.clipboard && window.isSecureContext) {
      navigator.clipboard.writeText(text)
        .then(() => alert("存档已生成并复制。你可以把它发送给自己，再在另一台设备导入。"))
        .catch(() => alert("存档已生成并选中，请手动复制文本。"));
    } else {
      alert("存档已生成并选中，请手动复制文本。");
    }
  } catch(e) {
    alert("存档已生成并选中，请手动复制文本。");
  }
}

function importSave() {
  const text = document.getElementById("saveText").value.trim();
  if (!text) {
    alert("请先在文本框粘贴存档内容。");
    return;
  }
  try {
    const parsed = JSON.parse(text);
    if (!parsed || !parsed.teams || !Array.isArray(parsed.logs)) {
      throw new Error("格式不正确");
    }
    if (!confirm("导入后会覆盖当前设备的存档，确定继续吗？")) return;
    state = {...structuredClone(defaults),...parsed};
    if (!Array.isArray(state.teams) || state.teams.length!==2) state.teams = [[],[]];
    if (state.playerSide !== 0 && state.playerSide !== 1) state.playerSide = 0;
    syncSelected();
    undoState = null;
    persist();
    renderRoster();
    renderPreview();
    renderSidePicker();
    if (state.game) {
      if (state.game.over) showResult();
      else showGame();
    } else {
      backToSetup();
    }
    alert("存档导入成功。");
  } catch(e) {
    alert("无法识别该存档，请检查是否完整复制。");
  }
}

function resetEverything() {
  if (!confirm("确定清除本设备的全部游戏数据吗？此操作无法撤销。建议先导出存档。")) {
    return;
  }
  clearTimeout(aiTimer);
  aiRunning = false;
  localStorage.removeItem(STORAGE_KEY);
  state = structuredClone(defaults);
  undoState = null;
  document.getElementById("saveText").value = "";
  backToSetup();
  persist();
}

  }
}
